### 3.2 `@oblivion/core/lib/index.js`

```js
import z from '@deepseek-ai/schemastery'
import { Config } from './config.js'
import { registerKnowledge } from './knowledge/index.js'
import { registerProfile } from './profile/index.js'
import { registerGraph } from './graph/index.js'
import { registerFeedback } from './feedback/index.js'
import { registerQaLoop } from './qa-loop/index.js'
import { registerPerspective } from './perspective/index.js'
import { registerTools } from './tools.js'
import { OBLIVION_SYSTEM_PROMPT } from './prompt.js'

export const name = 'oblivion-core'
export const inject = ['tools', 'llm', 'systemPrompt', 'session']   // ← 无 webServer
export { Config }

export function apply(ctx, config) {
  const knowledge = registerKnowledge(ctx, config)
  const profile = registerProfile(ctx, config)
  const graph = registerGraph(ctx, config, knowledge)
  const feedback = config.enableFeedback ? registerFeedback(ctx, config, profile) : null

  const deps = { knowledge, profile, graph, feedback }
  registerQaLoop(ctx, config, deps)
  if (config.enablePerspective) registerPerspective(ctx, config, deps)
  registerTools(ctx, deps)

  // ✅ 真实签名：{ name, order, text }
  ctx.effect(
    () => ctx.systemPrompt.section({
      name: 'oblivion:knowledge',
      order: 3150,                       // 紧跟 MCP_SERVERS:3100
      text: () => OBLIVION_SYSTEM_PROMPT,
    }),
    'oblivion: knowledge prompt',
  )

  // 供 http-bridge 通过 ctx.get('oblivion') 机会式取用
  ctx.provide?.('oblivion', deps) ?? ctx.set?.('oblivion', deps)
}
```

> 如果不想依赖 `ctx.provide`，退回 `ctx.oblivion = deps` + 在 `@deepseek-ai/cordis` 上做 module augmentation 声明该字段，避免 `as any`。

### 3.5 `@oblivion/http-bridge/lib/index.js`（修正后的正确写法）

```js
import { credentialRef } from '@deepseek-ai/dsh-credentials'
import { Config } from './config.js'
import { checkAuth, readToken } from './auth.js'
import { readBody } from './body.js'
import { handleMCP } from './mcp.js'

export const name = 'oblivion-http-bridge'
export const inject = ['webServer', 'credentials']
export { Config }

const PROTOCOL_VERSION = '2025-06-18'   // 与现役 shell host 对齐

const ALLOWED_ORIGINS = new Set([
  'chrome-extension://koealbifonogjljlppjngjnaejemdjhp',  // Chrome
  'chrome-extension://kbiaaejpjnpfcjaohkamimhpkoeaajcp',  // Edge
])

export function apply(ctx, config) {
  const ref = credentialRef(config.tokenEnv)

  const route = {
    kind: 'exact',                     // ✅ 必填，原设计漏了
    path: config.path,                 // '/oblivion/mcp'，无尾斜杠
    handler: async (req, res) => {
      const origin = req.headers.origin ?? ''
      const cors = ALLOWED_ORIGINS.has(origin)
        ? {
            'access-control-allow-origin': origin,
            'access-control-allow-methods': 'POST, OPTIONS',
            'access-control-allow-headers': 'authorization, content-type',
            'access-control-max-age': '600',
          }
        : {}

      if (req.method === 'OPTIONS') { res.writeHead(204, cors).end(); return }

      if (req.method !== 'POST') {
        res.writeHead(405, { allow: 'POST, OPTIONS', ...cors }).end()
        return
      }

      const token = await readToken(ctx, ref)
      if (!token) { res.writeHead(503).end('bridge token unavailable'); return }
      if (!checkAuth(req, token)) { res.writeHead(401).end('unauthorized'); return }

      let raw
      try { raw = await readBody(req, config.maxBodyBytes) }
      catch (e) {
        res.writeHead(e.status ?? 413).end(e.message ?? 'payload too large')
        return
      }

      const result = await handleMCP(ctx, raw, { ...config, protocolVersion: PROTOCOL_VERSION })
      if (result === null) { res.writeHead(204, cors).end(); return }

      res.writeHead(200, { 'content-type': 'application/json', ...cors })
      res.end(JSON.stringify(result))
    },
  }

  // ✅ 必须用 ctx.effect —— 重复注册会抛，HMR 重载会二次注册
  ctx.effect(() => ctx.webServer.register(route), `oblivion: ${config.path}`)
  ctx.logger.info(`[oblivion] bridge on http://127.0.0.1:${ctx.webServer.port}${config.path}`)
}
```

### 3.6 `lib/auth.js`

```js
const LOOPBACK = new Set(['127.0.0.1', '::1', '::ffff:127.0.0.1'])

export function checkAuth(req, token) {
  const ip = req.socket.remoteAddress ?? ''
  if (!LOOPBACK.has(ip)) return false
  return (req.headers['authorization'] ?? '') === `Bearer ${token}`
}

// 每请求解析，支持运行期换密钥（官方 webhook 适配器同款做法）
export async function readToken(ctx, ref) {
  const c = await ctx.credentials.resolve(ref)
  return c?.value ?? ''
}
```
